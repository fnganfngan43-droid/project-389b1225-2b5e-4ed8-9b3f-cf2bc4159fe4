import React, { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import { 
  Account, 
  AccountGroup, 
  Currency, 
  Governorate, 
  Voucher, 
  OpeningBalance, 
  Invoice,
  Settings,
  CurrencyExchange
} from '@/types/accounting';

interface AccountingContextType {
  // Settings
  settings: Settings;
  updateSettings: (settings: Settings) => void;
  
  // Login state
  isLoggedIn: boolean;
  setIsLoggedIn: (value: boolean) => void;
  
  // Accounts
  accounts: Account[];
  addAccount: (account: Account) => void;
  updateAccount: (account: Account) => void;
  deleteAccount: (id: string) => void;
  
  // Account Groups
  groups: AccountGroup[];
  addGroup: (group: AccountGroup) => void;
  updateGroup: (group: AccountGroup) => void;
  deleteGroup: (id: string) => void;
  
  // Currencies
  currencies: Currency[];
  addCurrency: (currency: Currency) => void;
  updateCurrency: (currency: Currency) => void;
  deleteCurrency: (id: string) => void;
  
  // Governorates
  governorates: Governorate[];
  addGovernorate: (governorate: Governorate) => void;
  updateGovernorate: (governorate: Governorate) => void;
  deleteGovernorate: (id: string) => void;
  
  // Vouchers
  vouchers: Voucher[];
  addVoucher: (voucher: Voucher) => void;
  updateVoucher: (voucher: Voucher) => void;
  deleteVoucher: (id: string) => void;
  
  // Opening Balances
  openingBalances: OpeningBalance[];
  addOpeningBalance: (balance: OpeningBalance) => void;
  updateOpeningBalance: (balance: OpeningBalance) => void;
  deleteOpeningBalance: (id: string) => void;
  
  // Invoices
  invoices: Invoice[];
  addInvoice: (invoice: Invoice) => void;
  updateInvoice: (invoice: Invoice) => void;
  deleteInvoice: (id: string) => void;
  
  // Currency Exchanges
  currencyExchanges: CurrencyExchange[];
  addCurrencyExchange: (exchange: CurrencyExchange) => void;
  updateCurrencyExchange: (exchange: CurrencyExchange) => void;
  deleteCurrencyExchange: (id: string) => void;
}

const AccountingContext = createContext<AccountingContextType | undefined>(undefined);

const defaultSettings: Settings = {
  userName: '',
  headerArabic: ['اسم الشركة', 'العنوان', 'رقم الهاتف'],
  headerEnglish: ['Company Name', 'Address', 'Phone Number'],
};

export function AccountingProvider({ children }: { children: ReactNode }) {
  // Load from localStorage or use defaults
  const loadFromStorage = <T,>(key: string, defaultValue: T): T => {
    const stored = localStorage.getItem(key);
    return stored ? JSON.parse(stored) : defaultValue;
  };

  const [settings, setSettings] = useState<Settings>(() => loadFromStorage('settings', defaultSettings));
  const [isLoggedIn, setIsLoggedIn] = useState(() => loadFromStorage('isLoggedIn', false));
  const [accounts, setAccounts] = useState<Account[]>(() => loadFromStorage('accounts', []));
  const [groups, setGroups] = useState<AccountGroup[]>(() => loadFromStorage('groups', []));
  const [currencies, setCurrencies] = useState<Currency[]>(() => loadFromStorage('currencies', []));
  const [governorates, setGovernorates] = useState<Governorate[]>(() => loadFromStorage('governorates', []));
  const [vouchers, setVouchers] = useState<Voucher[]>(() => loadFromStorage('vouchers', []));
  const [openingBalances, setOpeningBalances] = useState<OpeningBalance[]>(() => loadFromStorage('openingBalances', []));
  const [invoices, setInvoices] = useState<Invoice[]>(() => loadFromStorage('invoices', []));
  const [currencyExchanges, setCurrencyExchanges] = useState<CurrencyExchange[]>(() => loadFromStorage('currencyExchanges', []));

  // Save to localStorage whenever state changes
  useEffect(() => { localStorage.setItem('settings', JSON.stringify(settings)); }, [settings]);
  useEffect(() => { localStorage.setItem('isLoggedIn', JSON.stringify(isLoggedIn)); }, [isLoggedIn]);
  useEffect(() => { localStorage.setItem('accounts', JSON.stringify(accounts)); }, [accounts]);
  useEffect(() => { localStorage.setItem('groups', JSON.stringify(groups)); }, [groups]);
  useEffect(() => { localStorage.setItem('currencies', JSON.stringify(currencies)); }, [currencies]);
  useEffect(() => { localStorage.setItem('governorates', JSON.stringify(governorates)); }, [governorates]);
  useEffect(() => { localStorage.setItem('vouchers', JSON.stringify(vouchers)); }, [vouchers]);
  useEffect(() => { localStorage.setItem('openingBalances', JSON.stringify(openingBalances)); }, [openingBalances]);
  useEffect(() => { localStorage.setItem('invoices', JSON.stringify(invoices)); }, [invoices]);
  useEffect(() => { localStorage.setItem('currencyExchanges', JSON.stringify(currencyExchanges)); }, [currencyExchanges]);

  const value: AccountingContextType = {
    settings,
    updateSettings: setSettings,
    isLoggedIn,
    setIsLoggedIn,
    accounts,
    addAccount: (account) => setAccounts(prev => [...prev, account]),
    updateAccount: (account) => setAccounts(prev => prev.map(a => a.id === account.id ? account : a)),
    deleteAccount: (id) => setAccounts(prev => prev.filter(a => a.id !== id)),
    groups,
    addGroup: (group) => setGroups(prev => [...prev, group]),
    updateGroup: (group) => setGroups(prev => prev.map(g => g.id === group.id ? group : g)),
    deleteGroup: (id) => setGroups(prev => prev.filter(g => g.id !== id)),
    currencies,
    addCurrency: (currency) => setCurrencies(prev => [...prev, currency]),
    updateCurrency: (currency) => setCurrencies(prev => prev.map(c => c.id === currency.id ? currency : c)),
    deleteCurrency: (id) => setCurrencies(prev => prev.filter(c => c.id !== id)),
    governorates,
    addGovernorate: (gov) => setGovernorates(prev => [...prev, gov]),
    updateGovernorate: (gov) => setGovernorates(prev => prev.map(g => g.id === gov.id ? gov : g)),
    deleteGovernorate: (id) => setGovernorates(prev => prev.filter(g => g.id !== id)),
    vouchers,
    addVoucher: (voucher) => setVouchers(prev => [...prev, voucher]),
    updateVoucher: (voucher) => setVouchers(prev => prev.map(v => v.id === voucher.id ? voucher : v)),
    deleteVoucher: (id) => setVouchers(prev => prev.filter(v => v.id !== id)),
    openingBalances,
    addOpeningBalance: (balance) => setOpeningBalances(prev => [...prev, balance]),
    updateOpeningBalance: (balance) => setOpeningBalances(prev => prev.map(b => b.id === balance.id ? balance : b)),
    deleteOpeningBalance: (id) => setOpeningBalances(prev => prev.filter(b => b.id !== id)),
    invoices,
    addInvoice: (invoice) => setInvoices(prev => [...prev, invoice]),
    updateInvoice: (invoice) => setInvoices(prev => prev.map(i => i.id === invoice.id ? invoice : i)),
    deleteInvoice: (id) => setInvoices(prev => prev.filter(i => i.id !== id)),
    currencyExchanges,
    addCurrencyExchange: (exchange) => setCurrencyExchanges(prev => [...prev, exchange]),
    updateCurrencyExchange: (exchange) => setCurrencyExchanges(prev => prev.map(e => e.id === exchange.id ? exchange : e)),
    deleteCurrencyExchange: (id) => setCurrencyExchanges(prev => prev.filter(e => e.id !== id)),
  };

  return (
    <AccountingContext.Provider value={value}>
      {children}
    </AccountingContext.Provider>
  );
}

export function useAccounting() {
  const context = useContext(AccountingContext);
  if (!context) {
    throw new Error('useAccounting must be used within AccountingProvider');
  }
  return context;
}

function openPrintWindow(content: string) {
  const printWindow = window.open('', '_blank', 'width=800,height=600');
  if (printWindow) {
    printWindow.document.write(content);
    printWindow.document.close();
    printWindow.focus();
    setTimeout(() => {
      printWindow.print();
    }, 500);
  }
